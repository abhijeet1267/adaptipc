Reproducibility

# full experiment campaign (raw CSVs -> summaries -> figures)
./demo/adaptipc_lab.sh -runall
# showcase figures + dashboard data (from committed raw CSVs)
./scripts/generate_showcase.sh
# interactive dashboard
./scripts/start_website.sh   # http://localhost:8123/index.html
